# D1 23DoF Fine-tune Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Fine-tune the released 29-DoF SONIC checkpoint into a 23-DoF-hardware-compatible policy by physics-locking 6 missing joints in IsaacLab during training, while keeping deploy/ONNX dimensions at 29 (zero changes to C++ stack).

**Architecture:** Two-phase plan. Phase 0–1 install IsaacLab 2.3.2 and probe 4 spec-flagged unknowns. Phase 2 lands probe-independent code (constants module, reference-motion preprocessor, hydra wrapper). Phase 3 lands probe-dependent code (physics lock wrapper, action/obs/reward masks). Phase 4 runs smoke test, binary-searches `num_envs` for the 4060 8 GB, and runs the 2-day local training. Single hydra flag `+missing_dofs=23dof_hardware` toggles all 6 changes.

**Tech Stack:** IsaacLab 2.3.2, IsaacSim 5.x, Python 3.11, PyTorch ≥ 2.4, Hydra 1.3.2, TRL 0.28.0, conda

**Spec:** `D1_23DOF_DESIGN_2026-05-15.md` (project root)

**Repo root (reference for all paths):** `/media/woan/84a38787-1d4e-4ba7-892e-d1d90a009a8c/lgy/GR00T-WholeBodyControl`

```python
"""Generate 23-DoF-compatible reference motion library.

Zeroes the 6 IL columns listed in joint_constants.MISSING_23DOF_INDICES_IL in
the joint_pos / joint_vel arrays of every PKL under --source, writing to
--dest. Mirrors the deploy-side helper gear_sonic_deploy/reference/zero_23dof_columns.py
but operates on the training motion_lib PKLs instead of CSVs.

Usage:
    python -m gear_sonic.data_process.zero_23dof_motion_lib \
        --source data/motion_lib_bones_seed/robot_filtered \
        --dest   data/motion_lib_bones_seed/robot_filtered_23dof \
        --workers 16
"""
from __future__ import annotations

import argparse
import pickle
from concurrent.futures import ProcessPoolExecutor
from pathlib import Path

import numpy as np

from gear_sonic.utils.joint_constants import MISSING_23DOF_INDICES_IL


def zero_23dof_arrays_in_pkl(src: Path, dst: Path) -> None:
    with open(src, "rb") as f:
        data = pickle.load(f)
    for key in ("joint_pos", "joint_vel"):
        if key in data and isinstance(data[key], np.ndarray) and data[key].ndim == 2:
            data[key] = data[key].copy()
            data[key][:, MISSING_23DOF_INDICES_IL] = 0
    dst.parent.mkdir(parents=True, exist_ok=True)
    with open(dst, "wb") as f:
        pickle.dump(data, f)


def _worker(args):
    src, dst = args
    zero_23dof_arrays_in_pkl(src, dst)
    return src.name


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--source", type=Path, required=True)
    p.add_argument("--dest", type=Path, required=True)
    p.add_argument("--workers", type=int, default=8)
    args = p.parse_args()

    pkls = sorted(args.source.rglob("*.pkl"))
    print(f"[zero_23dof] {len(pkls)} PKLs in {args.source}")
    args.dest.mkdir(parents=True, exist_ok=True)

    jobs = [(p, args.dest / p.relative_to(args.source)) for p in pkls]
    with ProcessPoolExecutor(max_workers=args.workers) as ex:
        for i, name in enumerate(ex.map(_worker, jobs)):
            if i % 1000 == 0:
                print(f"  {i}/{len(jobs)} {name}")
    print(f"[zero_23dof] done. Output: {args.dest}")


if __name__ == "__main__":
    main()
```

```python
from typing import Sequence
import torch


def apply_missing_dof_mask(
    actions: torch.Tensor,
    missing_idx: Sequence[int] | None,
) -> torch.Tensor:
    """Zero the columns at `missing_idx` in `actions` (last dim).

    masked[..., list(missing_idx)] = 0.0
```

```python
"""Environment wrapper that physically locks the 6 missing-on-23DoF joints.

Each step, after physics steps, hard-resets the 6 joints' qpos to default
and qvel to 0. This makes the IsaacLab simulation state match a 23DoF
hardware variant where these joints physically do not exist. Without this
lock, the policy still sees the joints free-moving and learns to use them
for balance — see 23DOF_EVALUATION_2026-05-14.md for the prior failure.

Use case: D1 path (training-side fine-tune for 23DoF G1 hardware).
Activated by hydra config `missing_dofs.enabled: true`.
"""
from __future__ import annotations

from typing import Any

import torch
from isaaclab.envs import ManagerBasedRLEnv


class MissingDofsLockEnvWrapper(ManagerBasedRLEnv):
    """RL env that pins missing-DoF joints to their defaults each step."""

    def __init__(self, cfg, **kwargs):
        super().__init__(cfg, **kwargs)
        md_cfg = getattr(cfg, "missing_dofs", None)
        if md_cfg is None or not md_cfg.get("enabled", False):
            self._missing_lock_active = False
            return

        self._missing_lock_active = True
        self._missing_il_idx = list(md_cfg["indices_il"])
        # Cache joint defaults from the robot articulation
        robot = self.scene["robot"]
        self._missing_joint_ids_t = torch.tensor(
            self._missing_il_idx, device=self.device, dtype=torch.long
        )
        # qpos default — IsaacLab stores in articulation.data.default_joint_pos
        self._missing_qpos_default = robot.data.default_joint_pos[
            :, self._missing_joint_ids_t
        ].clone()
        self._missing_qvel_zero = torch.zeros_like(self._missing_qpos_default)

    def _apply_action(self) -> None:
        # 1. Apply the (already-action-masked) actions
        super()._apply_action()
        if not self._missing_lock_active:
            return
        # 2. Immediately overwrite the 6 missing joints' state to default
        robot = self.scene["robot"]
```
